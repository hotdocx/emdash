from pathlib import Path
import hashlib,json,os,shutil,subprocess,tempfile,time

laws={'emdash3_2_one_cat_native_snake_outer_zeros.lp':['one_cat_native_snake_kernel_pair_zero','one_cat_native_snake_cokernel_pair_zero'],
      'emdash3_2_one_cat_native_snake_inner_zeros.lp':['one_cat_native_snake_connecting_kernel_zero','one_cat_native_snake_cokernel_connecting_zero']}
record={'gc':'o=20,v=1024','memory_mib':2048,'seconds_per_target':90,'laws':laws,'checks':[],'new_assumptions':0,'active_sources_changed':False}

with tempfile.TemporaryDirectory(prefix='emdash-native-snake-opaque-laws-') as folder:
 stage=Path(folder);record['stage']=folder
 shutil.copy2(root/'lambdapi.pkg',stage/'lambdapi.pkg')
 for source in root.glob('*.lp'):shutil.copy2(source,stage/source.name)
 (stage/'examples').mkdir();target='examples/one_cat_native_snake_six_term_exactness.lp';shutil.copy2(root/target,stage/target)
 for file,names in laws.items():
  text=(stage/file).read_text()
  for name in names:
   start=text.index('symbol '+name);body=text.index('≔',start);header=text[start:body]
   assert 'IsNCat (cat_succ cat_zero)' in header and '@= (@Transf' in header,name
   end=text.index(';',body)+1
   text=text[:end]+'\nopaque '+name+';\n'+text[end:]
  (stage/file).write_text(text)
 record['source_sha256']={str(p.relative_to(stage)):hashlib.sha256(p.read_bytes()).hexdigest() for p in stage.rglob('*.lp')}
 for label,file in [('owner','emdash3_2_one_cat_native_snake_six_term_result.lp'),('reviewer',target)]:

  row={'file':file,'exit':result.returncode,'seconds':round(time.monotonic()-started,3),'log':str(log.relative_to(root))};record['checks'].append(row);print(json.dumps(row),flush=True)
  (root/'tmp/probes/nuh6c3_opaque_laws.json').write_text(json.dumps(record,indent=2)+'\n')
  if result.returncode:break
record['temporary_stage_removed']=not Path(record['stage']).exists()
(root/'tmp/probes/nuh6c3_opaque_laws.json').write_text(json.dumps(record,indent=2)+'\n')
